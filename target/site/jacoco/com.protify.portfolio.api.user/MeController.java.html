package com.protify.portfolio.api.user;

import com.protify.portfolio.api.dto.UserResponse;
import com.protify.portfolio.api.mapper.UserMapper;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RestController;

/**
 * The first real vertical slice (day-1-dev-C.md D1-C3): security filter -> resolver ->
 * controller -> mapper -> DTO. No business logic here, only delegation and mapping —
 * persistence types never cross this boundary, which is why this method returns
 * {@link UserResponse} and never touches whatever {@link CurrentUserResolver}'s real
 * implementation looks up internally.
 */
@RestController
public class MeController {

    private final CurrentUserResolver currentUserResolver;
    private final UserMapper userMapper;

    public MeController(CurrentUserResolver currentUserResolver, UserMapper userMapper) {
        this.currentUserResolver = currentUserResolver;
        this.userMapper = userMapper;
    }

    @GetMapping("/me")
    public UserResponse me() {
        return userMapper.toResponse(currentUserResolver.resolve());
    }
}
